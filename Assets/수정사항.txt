PlayerWeaponStatsRuntime 스크립트 수정

변수추가

[SerializeField] private float dashDurationSec = 0.2f; //대시 지속시간

[SerializeField] private float dashDamage = 5.0f; //대시 데미지

[SerializeField] private float boomerangDamage = 5.0f;

[SerializeField] private float boomerangMaxDistance = 6.0f;


아래쪽에 함수 추가

public float GetDashDurationSec()
{
	return dashDurationSec;
}

public float GetDashDamage()
{
	return dashDamage;
}

public float GetBoomerangDamage()
{
	return boomerangDamage;
}

public float GetBoomerangMaxDistance()
{
	return boomerangMaxDistance;
}


public void AddDashUpgrade(float durationSec, float damage) //지속시간과 데미지를 파라미터로 받음
{
	dashDurationSec += durationSec;
	dashDamage += damage;
}

public void AddBoomerangUpgrade(float damage, float maxDistance)
{
	boomerangMaxDistance += damage;
	boomerangMaxDistance += maxDistance;
}
